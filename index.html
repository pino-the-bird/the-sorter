the code it runs as

// Kiwi Shrine Note Sorter, version 1. built from the sorter parts on the internet cafe computer
const RULES = [
  { part: "people you know", does: "first", test: (n) => n.knows },
  { part: "asks you something", does: "first", test: (n) => n.question },
  { part: "with a picture", does: "most", aDay: 3, test: (n) => n.picture },
  { part: "what you've been into", does: "most", aDay: 3, test: (n) => n.into.length > 0 },
]
const REST = "back"
const A_DAY = 100

function has(n, words) { return words.some((w) => n.stems.includes(w) || n.pairs.includes(w)) }

// notes: the ones waiting, each as the town reads it. today: how many came in front of you today and were read, by part
function sort(notes, today) {
  const first = []
  const back = []
  const used = Object.assign({}, today.byPart)
  let room = A_DAY - today.read
  const placed = notes.map((n) => {
    const i = RULES.findIndex((r) => r.test(n))
    return { n, i: i === -1 ? RULES.length : i }
  }).sort((a, b) => a.i - b.i || b.n.waited - a.n.waited)
  for (const { n, i } of placed) {
    const rule = RULES[i]
    const by = rule ? rule.part : 'the rest'
    const does = rule ? rule.does : REST
    if (does === 'back') { back.push({ id: n.id, by }); continue }
    if (does === 'most' && (used[by] || 0) >= rule.aDay) { back.push({ id: n.id, by }); continue }
    if (room <= 0) { back.push({ id: n.id, by: 'a day' }); continue }
    used[by] = (used[by] || 0) + 1
    room--
    first.push({ id: n.id, by })
  }
  return { first, back }
}
